       ```python
       @model_validator(mode="after")
       def _validate_cors_origins(self) -> "Settings":
           """Fail closed on an unsafe production CORS origin (T-09-13).

           A trailing slash, a wildcard or an empty entry produces a valid
           string but never matches the browser Origin header, so the failure
           would only surface as a 400 preflight in production. Outside dev the
           process refuses to boot instead.
           """
           if self.ENV == "dev":
               return self

           origins = self.cors_origins_list
           if not origins or any(not origin for origin in origins):
               raise ValueError("CORS_ORIGINS é obrigatória fora de dev (T-09-13)")
           for origin in origins:
               if origin == "*":
                   raise ValueError(
                       "CORS_ORIGINS não aceita '*' fora de dev (T-09-13)"
                   )
               if origin.endswith("/"):
                   raise ValueError(
                       f"CORS_ORIGINS não pode ter barra final: {origin!r} (T-09-13)"
                   )
               if not origin.startswith(("http://", "https://")):
                   raise ValueError(
                       f"CORS_ORIGINS inválida (sem http/https): {origin!r} (T-09-13)"
                   )
           return self
       ```

       ```python
       """Production CORS contract (criterion 6 — T-09-13).

       The app must echo the configured origin exactly, refuse to echo a
       foreign origin, allow the report POST preflight, and refuse to boot
       outside dev with an unsafe CORS_ORIGINS value.
       """

       import uuid

       import pytest
       from pydantic import ValidationError

       from app.core.config import Settings, get_settings

       # Match the origin the app under test was actually built with, so the
       # test never depends on the developer's local .env file.
       CONFIGURED_ORIGIN = get_settings().cors_origins_list[0]


       def _settings(**kwargs) -> Settings:
           base = {
               "DATABASE_URL": "postgresql+asyncpg://user:pass@localhost:5432/nqu_test",
               "ADMIN_PASSWORD_HASH": "not-a-real-hash",
               "JWT_SECRET": "not-a-real-secret",
               "_env_file": None,
           }
           base.update(kwargs)
           return Settings(**base)


       async def test_configured_origin_is_echoed_exactly(client):
           resp = await client.get(
               "/api/v1/public/jobs",
               params={"limit": 1},
               headers={"Origin": CONFIGURED_ORIGIN},
           )
           assert resp.status_code == 200
           assert resp.headers.get("access-control-allow-origin") == CONFIGURED_ORIGIN


       async def test_foreign_origin_is_not_echoed(client):
           resp = await client.get(
               "/api/v1/public/jobs",
               params={"limit": 1},
               headers={"Origin": "https://evil.example"},
           )
           assert resp.status_code == 200
           assert "access-control-allow-origin" not in resp.headers


       async def test_preflight_allows_report_post(client):
           resp = await client.options(
               f"/api/v1/public/jobs/{uuid.uuid4()}/report",
               headers={
                   "Origin": CONFIGURED_ORIGIN,
                   "Access-Control-Request-Method": "POST",
                   "Access-Control-Request-Headers": "content-type",
               },
           )
           assert resp.status_code == 200
           assert resp.headers.get("access-control-allow-origin") == CONFIGURED_ORIGIN
           assert "POST" in resp.headers.get("access-control-allow-methods", "").upper()


       @pytest.mark.parametrize(
           "origin",
           ["", "*", "https://nao-quero-usar.vercel.app/", "nao-quero-usar.vercel.app"],
       )
       def test_prod_rejects_unsafe_cors_origins(origin: str):
           with pytest.raises(ValidationError, match="CORS_ORIGINS"):
               _settings(
                   ENV="prod",
                   CORS_ORIGINS=origin,
                   TURNSTILE_SECRET_KEY="z" * 35,
               )


       def test_prod_accepts_exact_https_origin_and_strips_spaces():
           settings = _settings(
               ENV="prod",
               CORS_ORIGINS=" https://nao-quero-usar.vercel.app , https://admin.example ",
               TURNSTILE_SECRET_KEY="z" * 35,
           )
           assert settings.cors_origins_list == [
               "https://nao-quero-usar.vercel.app",
               "https://admin.example",
           ]


       def test_dev_still_accepts_localhost_default():
           settings = _settings(ENV="dev")
           assert settings.cors_origins_list == ["http://localhost:3000"]
       ```

## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| browser→API | Cross-origin requests; the Origin header is untrusted input to CORSMiddleware |
| environment→process | `CORS_ORIGINS`, `ENV`, `SCHEDULER_ENABLED`, secrets loaded at boot |
| operator→production | Manual Render deploy with forward-only migrations and no schema rollback |
| repo→Render | `render.yaml` blueprint; secrets are dashboard-only (`sync: false`) |
| smoke gate→release decision | `verify-deploy.ps1` output is the release evidence |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-09-13 | Tampering / information disclosure | Production CORS | mitigate | `_validate_cors_origins` refuses to boot outside dev for empty, wildcard, trailing-slash or scheme-less origins; tests pin exact echo, foreign-origin denial and report preflight |
| T-09-14 | Denial of service | Render cold start | mitigate | `uv run --no-sync` on the start command; the build step is the only dependency sync, removing the dev-group re-install from every wake-up |
| T-09-15 | Repudiation | Scheduler silently disabled/enabled | mitigate | `SCHEDULER_ENABLED` is declared explicitly in `render.yaml` and documented; the cron catch-up remains idempotent and independent |
| T-09-16 | Spoofing / legal trust | Smoke gate accepting weak "intermedi" text | mitigate | Check 5 asserts `somos apenas meio de divulga` exactly |
