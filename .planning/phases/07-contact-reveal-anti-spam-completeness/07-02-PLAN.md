    1. Em `backend/app/api/v1/public.py`, adicionar as imports necessárias após as imports existentes:
      ```python
      from fastapi import APIRouter, Request, Depends, HTTPException, status, Body
      from slowapi import Limiter
      from slowapi.errors import RateLimitExceeded
      from app.core.config import get_settings, effective_turnstile_secret, DEV_TURNSTILE_SECRET
      ```

    2. Adicionar o router e o endpoint após as imports:
      ```python
      router = APIRouter()

      @router.post("/jobs/{job_id}/reveal-contact", response_model=dict)
      async def reveal_contact(
           job_id: str,
           request: Request,
           turnstile_response: str = Body(...),
           settings: Settings = Depends(get_settings),
      ):
          # Buscar job no banco - ANTES da verificação para verificar existência
          job = await get_session().get(Job, job_id)  # simplified - usar modelo real
          
          if not job:
              raise HTTPException(status_code=404, detail="Job not found")

          # Verificar Turnstile
          if settings.ENV == "dev":
              # Modo dev: default always-pass com chaves de teste
              if not turnstile_response or re.match(r"^[123]x[0-9A-Za-z]{31}AA$", turnstile_response):
                  # Contact revealed em dev
                  return {
                      "job_id": str(job.id),
                      "contact_email": job.contact_email,
                      "contact_phone": job.contact_phone,
                      "revealed": True,
                  }
              # Token inválido em dev - ainda assim return contact (D-12 flexibilidade)
              # OU retornar 403? A especificação diz "always-pass" para chaves 1x/2x/3x
              # Para tokens inválidos em dev, ser conservador e return 403
              raise HTTPException(status_code=403, detail="Turnstile verification failed")

          # Em prod, validação fail-closed
          secret = effective_turnstile_secret(settings)
          if not secret:
              raise HTTPException(status_code=500, detail="TURNSTILE_SECRET_KEY não configurada (D-11)")

          verify_data = {
              "secret": secret,
              "response": turnstile_response,
              "remoteip": resolve_client_ip(request),
          }
          async with httpx.AsyncClient() as client:
              resp = await client.post(
                  "https://challenges.cloudflare.com/turnstile/v0/siteverify",
                  data=verify_data,
                  timeout=settings.TURNSTILE_TIMEOUT_SECONDS or 10.0,
              )
          data = resp.json()

          if not data.get("success"):
              raise HTTPException(status_code=403, detail="Turnstile verification failed (D-11)")

          # Rate limit já foi aplicado pelo slowapi no nível do router
          # Se chegou aqui, verification passou
          return {
              "job_id": str(job.id),
              "contact_email": job.contact_email,
              "contact_phone": job.contact_phone,
              "revealed": True,
          }
      ```
