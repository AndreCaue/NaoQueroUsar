    ```ts
    const ADS_DISABLED_VALUES = new Set(["false", "0", "off", "no", "disabled"]);

    export function parseAdsEnabled(value: string | undefined): boolean {
      const normalized = value?.trim().toLowerCase();
      if (!normalized) return true; // D-01: absent/empty => show ads
      if (ADS_DISABLED_VALUES.has(normalized)) return false;
      return true; // any other value preserves current behavior
    }

    export function isAdsEnabled(): boolean {
      // Static property access required for Next.js NEXT_PUBLIC inlining in client bundles.
      return parseAdsEnabled(process.env.NEXT_PUBLIC_ADS_ENABLED);
    }

    export function parseDonationUrl(value: string | undefined): string | null {
      const normalized = value?.trim();
      if (!normalized) return null;
      try {
        const parsed = new URL(normalized);
        if (parsed.protocol !== "http:" && parsed.protocol !== "https:") return null;
        return parsed.toString();
      } catch {
        return null;
      }
    }

    export function getDonationUrl(): string | null {
      // Static property access required for Next.js NEXT_PUBLIC inlining in client bundles.
      return parseDonationUrl(process.env.NEXT_PUBLIC_DONATION_URL);
    }
    ```

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| Deploy/env config → browser bundle | `NEXT_PUBLIC_*` values are operator-configured and embedded in client bundle; treated as untrusted config input for navigation/render decisions |
| Public UI → external destination | Doações button may navigate to operator-configured external URL in a new tab |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-27-01 | Tampering / Information Disclosure (unsafe navigation) | `frontend/lib/publicEnv.ts`, `frontend/app/(public)/layout.tsx` | mitigate | `parseDonationUrl` accepts only absolute `http:`/`https:` via `new URL()`; `javascript:`, `data:`, ftp, relative, malformed return `null`; layout falls back to internal `<Link href="/doacoes">` (D-05). Unit tests cover unsafe schemes. |
| T-27-02 | Spoofing / Tampering (reverse-tabnabbing) | `frontend/app/(public)/layout.tsx` | mitigate | External donation anchor always includes `target="_blank"` and `rel="noopener noreferrer"` (D-04). Acceptance criteria grep for both attributes. |
| T-27-03 | Denial of Service (misconfiguration / broken UX) | `frontend/lib/publicEnv.ts`, `frontend/components/board/AdSlot.tsx`, `frontend/scripts/verify-ssr.ps1` | mitigate | Safe defaults: ads visible when flag absent/invalid value (D-01); donation falls back to `/doacoes` when invalid (D-05). No throw on bad env. verify-ssr becomes env-aware to avoid false CI failures. |
</threat_model>
