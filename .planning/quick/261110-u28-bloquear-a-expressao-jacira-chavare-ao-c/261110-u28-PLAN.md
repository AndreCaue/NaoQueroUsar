```python
"""Termos bloqueados na extração (admin paste) — regra única do backend.

Regra (espelhada em frontend/lib/blocklist.ts): case-insensitive, sem
acentos (NFKD) e com espaços/quebras de linha colapsados. A comparação é
por substring em qualquer ponto do texto.

Nunca logue nem ecoe o texto colado: só o termo bloqueado aparece na
mensagem, que é exibida ao admin que fez o paste.
"""

import re
import unicodedata

BLOCKED_TERM = "JACIRA CHAVARE"

BLOCKED_TERM_ERROR = (
    'Texto bloqueado — remova "JACIRA CHAVARE" antes de extrair.'
)

_WHITESPACE = re.compile(r"\s+")


def _fold(value: str) -> str:
    """Normaliza para comparar: NFKD, sem marcas, espaços colapsados, casefold."""
    decomposed = unicodedata.normalize("NFKD", value)
    without_marks = "".join(ch for ch in decomposed if not unicodedata.combining(ch))
    return _WHITESPACE.sub(" ", without_marks).strip().casefold()


_FOLDED_TERM = _fold(BLOCKED_TERM)


def is_blocked_text(text: str) -> bool:
    """True quando o texto contém o termo bloqueado (substring, qualquer caixa)."""
    if not text:
        return False
    return _FOLDED_TERM in _fold(text)
```

Não use regex construída a partir do texto do usuário e não use `re.sub` com padrão dinâmico: a comparação é `in` sobre strings já normalizadas (evita backtracking e injeção de padrão).

```ts
/**
 * Termos bloqueados na extração (admin paste) — regra única do frontend.
 * Espelha backend/app/services/extraction/blocklist.py: case-insensitive,
 * sem acentos (NFKD) e com espaços/quebras de linha colapsados. A comparação
 * é por substring em qualquer ponto do texto. Trocar o termo = trocar aqui e
 * no módulo Python, nada mais.
 */

export const BLOCKED_TERM = "JACIRA CHAVARE";

export const BLOCKED_TERM_ERROR = `Texto bloqueado — remova "${BLOCKED_TERM}" antes de extrair.`;

function fold(value: string): string {
  return value
    .normalize("NFKD")
    // Marcas combinantes latinas (U+0300–U+036F). NÃO use \p{M}: o tsconfig
    // do projeto tem target ES2017 e Unicode property escapes exigem ES2018.
    .replace(/[\u0300-\u036f]/g, "")
    .replace(/\s+/g, " ")
    .trim()
    .toLowerCase();
}

const FOLDED_TERM = fold(BLOCKED_TERM);

export function isBlockedText(text: string): boolean {
  return fold(text).includes(FOLDED_TERM);
}
```
